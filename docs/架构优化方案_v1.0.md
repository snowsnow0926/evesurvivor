## 1. 背景与目标

### 1.1 现状

项目当前存在以下架构问题：

| 优先级 | 问题 | 影响范围 |
|--------|------|----------|
| P1 | player.gd 与 game_manager.gd 状态大量重复，同步极易出错 | 战斗系统 |
| P1 | boss_void.gd 偏离 EnemyBase 基类，重复代码严重 | 敌人系统 |
| P2 | game_manager.gd 是 God Class（1108行），职责过多 | 整体架构 |
| P3 | 场景/资源重复 load()，未预加载 | 性能 |
| P3 | _apply_chapter_icon() 在三个敌人脚本中重复 | 代码维护 |
| P4 | Debug print 不统一 | 代码规范 |
| P4 | _build_polygon_from_path() 在 player.gd 和 enemy_base.gd 中重复 | 代码复用 |

### 1.2 优化目标

1. 消除状态重复，实现单一数据源
2. 建立清晰的继承体系，减少代码重复
3. 拆分 God Class，提高可维护性
4. 统一代码规范，便于后续迭代
5. 优化性能，减少不必要的资源加载

#### 2.1.2 PlayerStats 设计

```gdscript
class_name PlayerStats
extends Resource

## 单一数据源：管理玩家所有战斗属性
## Player 和 GameManager 共享此实例，不再需要手动同步

signal hp_changed(hp: int, max_hp: int)
signal shield_changed(shield: float, max_shield: float)
signal stats_changed()  # 任意属性变化时触发，用于刷新 UI

# === 生命与护盾 ===
@export var hp: int = 100:
    set(v):
        hp = clampi(v, 0, max_hp)
        hp_changed.emit(hp, max_hp)
        stats_changed.emit()

@export var max_hp: int = 100:
    set(v):
        max_hp = v
        hp_changed.emit(hp, max_hp)
        stats_changed.emit()

@export var shield: float = 50.0:
    set(v):
        shield = clampf(v, 0.0, shield_max)
        shield_changed.emit(shield, shield_max)
        stats_changed.emit()

@export var shield_max: float = 50.0:
    set(v):
        shield_max = v
        shield_changed.emit(shield, shield_max)
        stats_changed.emit()

@export var shield_regen: float = 4.0
@export var shield_regen_timer: float = 0.0

# === 移动与生存 ===
@export var move_speed: float = 320.0
@export var dodge: float = 0.1
@export var lifesteal: float = 0.0
@export var lifesteal_timer: float = 0.0

# === 暴击 ===
@export var crit_rate: float = 0.05
@export var crit_mult: float = 1.5

# === 武器通用 ===
@export var damage: float = 15.0
@export var xp_boost: float = 1.0

# === 导弹系 ===
@export var missile_speed: float = 600.0
@export var missile_range: float = 500.0
@export var missile_splash_radius: float = 0.0
@export var missile_splash_count: int = 0
@export var missile_burst_interval: float = 0.1
@export var spread_count: int = 1
@export var spread_angle: float = 6.0

# === 加农炮系 ===
@export var cannon_fire_interval: float = 1.2
@export var cannon_damage: float = 25.0
@export var cannon_pierce_count: int = 1
@export var cannon_explode_chance: float = 0.0
@export var cannon_bloodthirst: int = 0
@export var cannon_rush_level: int = 0
@export var cannon_vengeance_level: int = 0

# === 磁轨炮系 ===
@export var railgun_damage: float = 30.0
@export var railgun_speed: float = 1000.0
@export var railgun_range: float = 400.0
@export var railgun_fire_interval: float = 0.6
@export var railgun_crit_bonus: float = 0.0
@export var railgun_multi_count: int = 1
@export var railgun_burst_interval: float = 0.1

# === 激光系 ===
@export var laser_damage: float = 12.0
@export var laser_duration: float = 2.0
@export var laser_width: float = 16.0
@export var laser_fire_interval: float = 2.5
@export var laser_shield_mult: float = 1.0

# === 被动技能等级 ===
@export var silent_hunter_level: int = 0

# === 临时状态（不持久化）===
var is_moving: bool = false
var is_stationary: bool = false
var is_injured: bool = false
var injured_timer: float = 0.0
const INJURED_DURATION: float = 2.0

func _init() -> void:
    hp_changed.connect(_on_hp_changed)
    shield_changed.connect(_on_shield_changed)

func _on_hp_changed(current: int, maximum: int) -> void:
    if current <= 0:
        pass  # 死亡逻辑由调用方处理

func _on_shield_changed(current: float, maximum: float) -> void:
    pass  # 可扩展护盾破碎特效

func reset() -> void:
    var mh = max_hp
    var sm = shield_max
    var sr = shield_regen
    var ms = move_speed
    var d = dodge
    var cr = crit_rate
    var cm = crit_mult
    var ls = lifesteal
    var xb = xp_boost

    self = _new_instance()
    max_hp = mh
    shield_max = sm
    shield_regen = sr
    move_speed = ms
    dodge = d
    crit_rate = cr
    crit_mult = cm
    lifesteal = ls
    xp_boost = xb

static func _new_instance() -> PlayerStats:
    return new()

func apply_upgrade(upgrade_id: String) -> void:
    match upgrade_id:
        "damage":
            damage *= 1.2
        "shield_max":
            shield_max += 30.0
            shield = shield_max
        "fire_coverage":
            spread_count += 1
        "shield_regen":
            shield_regen *= 1.5
        "silent_hunter":
            silent_hunter_level += 1
        "precision_kill":
            missile_range += 100.0
        "cannon_bloodthirst":
            cannon_bloodthirst += 1
        "cannon_rush":
            cannon_rush_level += 1
        "cannon_vengeance":
            cannon_vengeance_level += 1
        "railgun_damage":
            railgun_damage *= 1.2
        "railgun_crit":
            railgun_crit_bonus += 0.1
        "railgun_multi":
            railgun_multi_count += 1
        "laser_duration":
            laser_duration *= 1.2
        "laser_width":
            laser_width *= 1.2
        "laser_shield":
            laser_shield_mult += 0.2
    stats_changed.emit()

func update_regen(delta: float) -> void:
    shield_regen_timer += delta
    if shield_regen_timer >= 1.0:
        shield_regen_timer = 0.0
        shield = minf(shield + shield_regen, shield_max)

    if lifesteal > 0.0:
        lifesteal_timer += delta
        if lifesteal_timer >= 1.0:
            lifesteal_timer = 0.0
            hp = mini(hp + int(lifesteal), max_hp)

func get_fire_interval_missile(base_interval: float) -> float:
    var mult = 1.0
    if is_stationary and silent_hunter_level > 0:
        mult = pow(0.85, float(silent_hunter_level))
    if is_moving and cannon_rush_level > 0:
        mult *= (1.0 - 0.15 * float(cannon_rush_level))
    return base_interval * mult

func get_fire_interval_cannon(base_interval: float) -> float:
    var interval = base_interval
    if is_moving and cannon_rush_level > 0:
        interval *= (1.0 - 0.15 * float(cannon_rush_level))
    if is_injured and cannon_vengeance_level > 0:
        interval *= (1.0 - 0.15 * float(cannon_vengeance_level))
    return interval
```

### Phase 2: BossVoid 统一到 EnemyBase

**目标**: 让 BossVoid 正确继承 EnemyBase，消除重复代码

#### 2.2.1 当前问题

BossVoid 重写了以下已在 EnemyBase 中实现的方法：

| 方法 | EnemyBase | BossVoid | 差异 |
|------|-----------|----------|------|
| `_spawn_damage_number()` | 有 | 有 | 字体大小/颜色不同 |
| `_spawn_death_effect()` | 有 | 有 | 粒子数量/规模不同 |
| `_start_hit_flash()` | 有 | 有 | 闪光强度不同 |
| `_update_hp_bar()` | 有 | 有 | HP条尺寸不同 |
| `setup_enemy()` | 有 | `setup_boss()` | 接口不统一 |

在 EnemyBase 中增加配置钩子，BossVoid 通过配置而非重写来实现差异：

var _death_particle_velocity_max: float = 150.0

# 命中闪光配置

### Phase 3: 场景预加载

**目标**: 用 `preload()` 替代运行时 `load()`，提升性能

#### 2.3.1 当前问题

```gdscript
# player.gd 第523行 — 每次发射都 load()
var missile_scene = load(missile_scene_path)
var missile = missile_scene.instantiate()

# game_manager.gd 第603行 — 每次生成敌人都 load()

在类级别用 `const` + `preload()` 预加载所有场景：

func _spawn_single_missile(target_pos: Vector2) -> void:
    var missile = _SCENE_MISSILE.instantiate()  # 直接实例化
    # ...

# === game_manager.gd 新增 ===

const _SCENE_MELEE: PackedScene = preload("res://scenes/EnemyMelee.tscn")
const _SCENE_SENTRY: PackedScene = preload("res://scenes/EnemySentry.tscn")

func _spawn_enemy() -> void:
    match enemy_path:
        ENEMY_MELEE_PATH:
            var enemy = _SCENE_MELEE.instantiate()  # 直接实例化
            # ...
```

### Phase 4: 敌人图标应用上移到基类

**目标**: 将 `_apply_chapter_icon()` 上移至 EnemyBase，消除3个敌人脚本中的重复

#### 2.4.1 当前问题

三个敌人脚本中完全相同的 switch-case（每个约40行）：

```gdscript
# enemy_melee.gd, enemy_sentry.gd, enemy_raven.gd 各有一份
func _apply_chapter_icon() -> void:
    var cid: int = ...
    match cid:
        2: var entry = ShipIconGenerator.get_entry(...)
            # ... 20行
        3: var entry = ...
            # ... 20行
        4: ...
        5: ...
```

#### 2.4.2 解决方案

在 EnemyBase 中实现通用的图标映射：

```gdscript
# === EnemyBase.gd 新增 ===

const _CHAPTER_ICON_MAP: Dictionary = {
    "melee": {
        2: "enemy_melee_cruiser",
        3: "enemy_melee_battlecruiser",
        4: "enemy_melee_battleship",
        5: "enemy_melee_dreadnought",
    },
    "sentry": {
        2: "enemy_sentry_cruiser",
        3: "enemy_sentry_battlecruiser",
        4: "enemy_sentry_battleship",
        5: "enemy_sentry_dreadnought",
    },
    "raven": {
        2: "enemy_raven_cruiser",
        3: "enemy_raven_battlecruiser",
        4: "enemy_raven_battleship",
        5: "enemy_raven_dreadnought",
    },
}

var _enemy_category_prefix: String = "melee"  # 子类设置

func _apply_chapter_icon() -> void:
    var cid: int = _chapter_id_override if _chapter_id_override > 0 else current_chapter_id
    var map = _CHAPTER_ICON_MAP.get(_enemy_category_prefix, {})
    var icon_id: String = map.get(cid, "")

    if icon_id.is_empty():
        return

    var entry: ShipIconGenerator.IconEntry = ShipIconGenerator.get_entry(
        ShipIconGenerator.Category.ENEMY, icon_id
    )
    if entry == null:
        return

    _icon_tex = entry.get_texture()
    if ship_sprite != null:
        ship_sprite.texture = _icon_tex
        ship_sprite.material = _tint_mat
        ship_sprite.visible = true
    if polygon != null:
        polygon.visible = false

# === enemy_melee.gd 修改后 ===
func _ready() -> void:
    _base_tint = Color(1.0, 0.2, 0.2)
    _icon_id = "enemy_melee"
    _enemy_category_prefix = "melee"
    tonnage = "destroyer"
    super()
    _death_particle_color = Color(1.0, 0.2, 0.2, 1.0)
    # _apply_chapter_icon() 由基类在 setup_enemy 中调用
```

#### 2.5.3 game_manager.gd 重构后架构

```gdscript
extends Node2D

const DEBUG := false

# === 子系统 ===
var spawn_manager: SpawnManager
var upgrade_system: UpgradeSystem
var difficulty_scaler: DifficultyScaler
var loot_system: LootSystem
var player_stats: PlayerStats  # Phase 1

# === 核心游戏状态 ===
var is_game_over: bool = false
var is_paused: bool = false
var is_upgrading: bool = false
var player_level: int = 1
var current_xp: float = 0.0
var xp_to_next_level: float = 10.0
var kill_count: int = 0
var total_kills: int = 0

# === 信号 ===
signal player_dead
signal upgrade_requested
signal game_paused(is_paused: bool)
signal game_ended(reason: String)

func _ready() -> void:
    _init_subsystems()
    _spawn_player()

func _init_subsystems() -> void:
    player_stats = PlayerStats.new()
    spawn_manager = SpawnManager.new(self, player_stats)
    upgrade_system = UpgradeSystem.new(player_stats)
    difficulty_scaler = DifficultyScaler.new()
    loot_system = LootSystem.new(self)

    upgrade_system.upgrade_applied.connect(_on_upgrade_applied)

func _process(delta: float) -> void:
    if is_game_over or is_paused or is_upgrading:
        return

    player_stats.update_regen(delta)
    difficulty_scaler.update_combo(delta)
    spawn_manager.update_spawning(delta)
    _notify_hud_update()

func _on_upgrade_applied() -> void:
    _notify_hud_update()
    if player and is_instance_valid(player):
        player.sync_from_player_stats(player_stats)
```

### 6.2 新发现问题

#### P0 — `enemy_sentry.gd` 的 `_physics_process` 覆盖问题（已确认为非 bug）

```gdscript
# enemy_sentry.gd 第 69 行
func _physics_process(delta: float) -> void:
	super._physics_process(delta)

func _process_combat(delta: float) -> void:
```

经分析确认：子类重写 `_physics_process` 并调用 `super._physics_process()` 后即结束——这是正常的。基类 `_physics_process` 内部调用 `_process_combat()`，子类各自的 `_process_combat` 实现通过基类统一调用，不存在重复执行。

**结论**：无需修改。各敌人子类写法虽然不一致，但 Godot 的继承调用链正确。

#### P1 — `player.gd` 每帧大量 `print()` debug 输出

```gdscript
# player.gd
if _physics_tick_counter % 60 == 0:
	print("[Player] _physics_process tick=", ...)  # 第 286 行

if _physics_tick_counter % 60 == 0:
	print("[Player] _update_firing: active_weapons=", ...)  # 第 370 行

# 武器分支内也有大量 debug print（第 477-496 行）
print("[Player] _fire_weapon: weapon=", ...)
print("[Player] match: MISSILE branch")
print("[Player] match: CANNON branch")
...
```

每 60 帧（约每秒 1 次）打印大量状态信息，武器发射分支内的 debug print 在每次开火时触发。游戏运行时将严重刷屏，必须清理或统一用 `DEBUG` 标志包裹。

#### P1 — `BossVoid` 图标系统三处操作同一对象

```
BossVoid._ready()      → super._ready()         → 基类 set_enemy_icon() 设置 _icon_tex
BossVoid._ready()      → setup_icon()            → 再次设置 _icon_tex（覆盖）
BossVoid.setup_boss()  → _setup_tonnage_icon()  → 第三次设置 _icon_tex（覆盖）
```

三处都在操作同一个 `_icon_tex` 和 `ship_sprite.texture`，流程冗余且容易出错。应在 `setup_boss()` 中统一处理，移除 `_ready()` 中的多余调用。
